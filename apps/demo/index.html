Get started

Quick start

Both read /photos/lake.jpg from the image server at /_img (@lazy-ease/server).

React

npm i @lazy-ease/react @lazy-ease/core thumbhash

import * as thumbhash from 'thumbhash'
import { detectAvif } from '@lazy-ease/core'
import { thumbhashDecoder } from '@lazy-ease/core/thumbhash'
import { Image, LazyEaseProvider } from '@lazy-ease/react'

const decoder = thumbhashDecoder(thumbhash)

export default function App() {
  return (
    <LazyEaseProvider thumbhash={decoder} detectFormat={detectAvif}>
      <Image src="/photos/lake.jpg" alt="Lake at dawn"
        width={320} height={240} srcWidth={2400} srcHeight={1800}
        placeholder="thumbhash:IqYJLYhclndfmHd7d4iIdY8wygik" />
    </LazyEaseProvider>
  )
}

HTML and a script

npm i @lazy-ease/vanilla, then copy lazy-ease.css and lazy-ease.min.js from its dist/ to /assets.

<link rel="stylesheet" href="/assets/lazy-ease.css">
<script defer src="/assets/lazy-ease.min.js"></script>

<section data-le-auto>
  <span data-le style="display:block;max-width:640px;aspect-ratio:4/3;background-color:#5b6c8f">
    <img alt="Lake at dawn" width="640" height="480"
      style="display:block;width:100%;height:100%;object-fit:cover"
      src="data:image/gif;base64,R0lGODlhAQABAAD/ACwAAAAAAQABAAACADs="
      data-le-url="/_img/photos/lake.jpg?fmt=webp&amp;q=75&amp;w=640">
  </span>
</section>
